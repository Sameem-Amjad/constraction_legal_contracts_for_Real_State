<h2>21. Résolution des différends</h2>
<p>En cas de différend survenant dans le cadre du présent Contrat, les parties adhèrent à la procédure de résolution suivante :</p>
<h3>21.1. Négociation</h3>
<p>Les parties tentent d&#x27;abord de résoudre le différend par la négociation directe, dans un délai de trente (30) jours à compter de la naissance du différend.</p>
<h3>21.2. Médiation</h3>
<p>Si le différend n&#x27;est pas résolu par la négociation, les parties conviennent de soumettre le différend à la médiation. Un médiateur accrédité au Québec sera choisi d&#x27;un commun accord dans les quinze (15) jours suivant la fin des négociations. Si les parties ne peuvent s&#x27;entendre sur le choix d&#x27;un médiateur, chaque partie propose un médiateur et ces médiateurs choisissent conjointement un troisième médiateur neutre pour mener la médiation.</p>
<h3>21.3. Calendrier de la médiation</h3>
<p>La médiation commence dans les trente (30) jours suivant la sélection du médiateur et se termine dans les soixante (60) jours, sauf si les parties en conviennent autrement par écrit.</p>
<h3>21.4. Coûts de la médiation</h3>
<p>Les coûts de la médiation sont partagés à parts égales entre les parties, sauf décision contraire du médiateur.</p>
<h3>21.5. Litiges</h3>
<p>Si la médiation ne permet pas de résoudre le différend, l&#x27;une ou l&#x27;autre des parties peut engager une procédure judiciaire. Les parties reconnaissent irrévocablement la compétence des tribunaux de l&#x27;arrondissement judiciaire où se trouve le Site du Projet.</p>
<p>Ces procédures assurent que les différends sont traités de manière cohérente et efficace, conformément à l&#x27;esprit de professionnalisme et de coopération voulu par les deux parties.</p>

// ===== Helper: Render ONE HTML string to a PDF blob using html2pdf =====
// Each invocation renders its own offscreen canvas, so long contracts that exceed
// the browser's ~16384px canvas cap can be handled by splitting the HTML and
// calling this twice (see generatePdfBlob below).
async function renderPartToPdfBlob(htmlString, fileName) {
  if (!window.html2pdf) throw new Error('html2pdf library failed to load');

  var container = document.createElement('div');
  container.style.cssText = 'display: block !important; align-items: start !important; justify-content: start !important; min-height: 0 !important; margin: 0 !important; width: 7.3in; background: #fff; padding: 0; font-family: "Times New Roman", serif; font-size: 12pt; line-height: 1.6; color: #000; text-align: left; box-sizing: border-box; overflow-wrap: break-word;';

  var style = document.createElement('style');
  style.textContent = `
      * { box-sizing: border-box; }
      h1 { font-size: 16pt; text-align: center; margin: 0 0 20px 0; font-weight: bold !important; page-break-after: avoid; page-break-inside: avoid; }
      h2 { font-size: 13pt; margin: 16px 0 8px; color: #000; page-break-after: avoid; page-break-inside: avoid; font-weight: bold !important; }
      h3 { font-size: 12pt; margin: 12px 0 6px; color: #000; page-break-after: avoid; page-break-inside: avoid; font-weight: bold !important; }
      p { margin: 8px 0; text-align: justify; color: #000; page-break-inside: avoid; }
      ul { margin: 8px 0 8px 20px; padding: 0; page-break-inside: avoid; }
      li { margin: 4px 0; page-break-inside: avoid; }
      .indented { padding-left: 20px; }
    `;
  container.appendChild(style);

  var contentDiv = document.createElement('div');
  contentDiv.innerHTML = htmlString;
  container.appendChild(contentDiv);

  // Fetch each logo using the authenticated Supabase client (bypasses CORS)
  var images = container.querySelectorAll('img');
  for (var i = 0; i < images.length; i++) {
    var img = images[i];
    if (img.src && img.src.includes('/logos/')) {
      try {
        var filePath = img.src.split('/logos/')[1].split('?')[0];
        var { data, error } = await _supabase.storage.from('logos').download(filePath);
        if (error || !data) throw error;
        var base64 = await new Promise((resolve, reject) => {
          var reader = new FileReader();
          reader.onloadend = () => resolve(reader.result);
          reader.onerror = reject;
          reader.readAsDataURL(data);
        });
        await new Promise((resolve) => {
          img.onload = resolve;
          img.onerror = resolve;
          img.src = base64;
        });
      } catch (e) {
        console.warn('Logo download failed.', e);
        if (img.parentNode && img.parentNode.tagName === 'DIV') img.parentNode.remove();
        else img.remove();
      }
    }
  }

  var opts = {
    margin: [0.6, 0.5, 0.6, 0.5],
    filename: fileName,
    pagebreak: { mode: ['css', 'legacy'], avoid: ['h1', 'h2', 'h3', 'p', 'li', 'ul'] },
    image: { type: 'jpeg', quality: 0.95 },
    // Each half renders its own canvas — we can afford scale 1.5 again since the
    // canvas height per half is now well under the 16384px browser cap.
    html2canvas: {
      scale: 1.5,
      useCORS: true,
      letterRendering: true,
      scrollY: 0,
      scrollX: 0
    },
    jsPDF: { unit: 'in', format: 'letter', orientation: 'portrait' }
  };

  return window.html2pdf().set(opts).from(container).output('blob');
}

// ===== Generate PDF blob from full contract HTML =====
// Splits on the `.lang-break` marker (EN/FR divider) into two halves, renders each
// half to its own PDF, then merges with pdf-lib. Single-language contracts skip the
// merge and render directly. Required for long bilingual contracts (20-30+ pages)
// that exceed the browser's ~16384px offscreen canvas cap in a single pass.
async function generatePdfBlob(htmlString, fileName) {
  if (!window.html2pdf) throw new Error('html2pdf library failed to load');

  var splitIdx = htmlString.indexOf('<div class="lang-break"');
  if (splitIdx < 0) {
    // No bilingual divider — render as one pass
    return renderPartToPdfBlob(htmlString, fileName);
  }

  // Strip `page-break-before: always` from the lang-break div when it's the start
  // of the second half — each half is already its own PDF, the forced break would
  // just produce a blank leading page.
  var primaryHtml = htmlString.substring(0, splitIdx);
  var secondaryHtml = htmlString.substring(splitIdx).replace(/page-break-before:\s*always;?\s*/i, '');

  var [blob1, blob2] = await Promise.all([
    renderPartToPdfBlob(primaryHtml, fileName),
    renderPartToPdfBlob(secondaryHtml, fileName)
  ]);

  // Merge the two PDFs page-by-page using pdf-lib
  if (!window.PDFLib) {
    // Give the loader a moment if the script hasn't finished loading yet
    await new Promise(r => setTimeout(r, 500));
    if (!window.PDFLib) throw new Error('pdf-lib failed to load');
  }

  var { PDFDocument } = window.PDFLib;
  var merged = await PDFDocument.create();
  for (var blob of [blob1, blob2]) {
    var buf = await blob.arrayBuffer();
    var src = await PDFDocument.load(buf);
    var copied = await merged.copyPages(src, src.getPageIndices());
    copied.forEach(function (p) { merged.addPage(p); });
  }
  var mergedBytes = await merged.save();
  return new Blob([mergedBytes], { type: 'application/pdf' });
}

  var authHeader = (token && token.access_token) ? 'Bearer ' + token.access_token : 'Bearer ' + SB_CONFIG.anonKey;
  // newContractId was already generated at the top of generate() and stored in APP._lastContractId

  // Client / Contractor party mapping for the database (CC form)
  var dbClientName, dbClientEmail, dbClientPhone, dbClientAddress, dbClientCity, dbClientPostal;
  var dbCtrName, dbCtrRbq, dbCtrEmail, dbCtrPhone, dbCtrAddress, dbCtrCity, dbCtrPostal;

  if (APP.role === 'role1') { // User is the Client; counterparty is the Contractor (ctr fields)
    dbClientName    = getPN('self');
    dbClientEmail   = d.selfEM || d.selfIE || d.selfOE || '';
    dbClientPhone   = d.selfPH || d.selfIP || d.selfOT || '';
    dbClientAddress = d.selfHO || d.selfIA || d.selfOA || '';
    dbClientCity    = d.selfHC || d.selfIC || d.selfOC || '';
    dbClientPostal  = d.selfHP || d.selfIPC || d.selfOP || '';
    dbCtrName       = getPN('ctr');
    dbCtrRbq        = d.ctrRBQ || '';
    dbCtrEmail      = d.ctrEM || d.ctrIE || d.ctrOE || '';
    dbCtrPhone      = d.ctrPH || d.ctrIP || d.ctrOT || '';
    dbCtrAddress    = d.ctrHO || d.ctrIA || d.ctrOA || '';
    dbCtrCity       = d.ctrHC || d.ctrIC || d.ctrOC || '';
    dbCtrPostal     = d.ctrHP || d.ctrIPC || d.ctrOP || '';
  } else { // User is the Contractor (self fields); counterparty is the Owner/Client (ow* fields)
    dbClientName    = getOwN();
    dbClientEmail   = d.owRE || d.owIE || d.owOE || '';
    dbClientPhone   = d.owRP || d.owIPH || d.owOT || '';
    dbClientAddress = d.owHO || d.owIA || d.owOA || '';
    dbClientCity    = d.owHC || d.owIC || d.owOC || '';
    dbClientPostal  = d.owHP || d.owIPC || d.owOP || '';
    dbCtrName       = getPN('self');
    dbCtrRbq        = d.selfRBQ || '';
    dbCtrEmail      = d.selfEM || d.selfIE || d.selfOE || '';
    dbCtrPhone      = d.selfPH || d.selfIP || d.selfOT || '';
    dbCtrAddress    = d.selfHO || d.selfIA || d.selfOA || '';
    dbCtrCity       = d.selfHC || d.selfIC || d.selfOC || '';
    dbCtrPostal     = d.selfHP || d.selfIPC || d.selfOP || '';
  }

  var fileName = 'contract_' + newContractId + '.pdf';
  var storagePath = (userId ? userId : 'guests') + '/' + fileName;

  generatePdfBlob(fullContractHtml, fileName)
    .then(function(pdfBlob) {
      return fetch(SB_CONFIG.url + '/storage/v1/object/contracts/' + storagePath, {
        method: 'POST',
        headers: { 'apikey': SB_CONFIG.anonKey, 'Authorization': authHeader, 'Content-Type': 'application/pdf', 'x-upsert': 'true' },
        body: pdfBlob
      }).then(r => r.ok ? storagePath : null);
    })
    .then(function(savedPath) {
      var contractRecord = { 
        id:                  newContractId,
        user_id:             userId,
        contract_type:       'client-contractor',
        client_name:         dbClientName || null,
        client_address:      dbClientAddress || null,
        client_city:         dbClientCity || null,
        client_postal:       dbClientPostal || null,
        client_email:        userEmail || null, 
        client_phone:        dbClientPhone || null,
        contractor_name:     dbCtrName || null,
        contractor_rbq:      dbCtrRbq || null,
        contractor_address:  dbCtrAddress || null,
        contractor_city:     dbCtrCity || null,
        contractor_postal:   dbCtrPostal || null,
        contractor_email:    dbCtrEmail || null,
        contractor_phone:    dbCtrPhone || null,
        project_site:        d.siteAddr || null,
        project_city:        d.siteCity || null,
        project_postal:      d.sitePostal || null,
        project_description: d.workDesc || null,
        contract_price:      d.totalPrice ? parseFloat(d.totalPrice) : null,
        status:              isPro ? 'paid' : 'generated', 
        pdf_path:            savedPath || null,
        metadata: {
          project_name:      d.projectName || '',
          language:          APP.lang,
          role:              APP.role,
          form_type:         FORM_TYPE,
          payment_method:    d.payMethod || '',
          start_date:        d.startDate || '',
          end_date:          d.endType === 'specific' ? (d.endDate || '') : ((d.endNum || '') + ' ' + (d.endUnit || '')),
          sign_date:         d.signDate || '',
          warranty_months:   d.warMon || '12',
          holdback:          d.holdback || 'no',
          holdback_pct:      d.hbPct || '',
          insurance_amount:  d.insAmt || '',
          bond:              d.bond || 'no',
          bond_pct:          d.bondPct || '',
          late_interest:     d.lateInt || '2',
          escalation:        d.escalation || 'no',
          escalation_pct:    d.escPct || '',
          recovery_penalty:  d.recPen || 'no',
          material_provider: d.matProv || '',
          extra_clauses:     d.extraCl || ''
        } 
      };
      return fetch(SB_CONFIG.url + '/rest/v1/contracts', {
        method: 'POST',
        headers: { 'apikey': SB_CONFIG.anonKey, 'Authorization': authHeader, 'Content-Type': 'application/json', 'Prefer': 'return=minimal' },
        body: JSON.stringify(contractRecord)
      }).then(r => { return { success: r.ok, storagePath: savedPath }; });
    })
    .then(function(data) {
      if (data.success && isPro && data.storagePath) {
        var emailFirstName = clientName ? clientName.split(' ')[0] : 'User';
        fetch(SB_CONFIG.url + '/functions/v1/send-contract', {
          method: 'POST',
          headers: { 'Authorization': 'Bearer ' + SB_CONFIG.anonKey, 'Content-Type': 'application/json' },
          body: JSON.stringify({ user_email: userEmail, first_name: emailFirstName, file_path: data.storagePath, language: APP.lang, contract_id: newContractId })
        });
      }
    })
    .catch(e => console.error('Save contract error:', e));
};
